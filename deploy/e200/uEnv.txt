maxcpus=2
nyxload=if fatload mmc 0 0x10000000 nyx.bit; then fpga loadb 0 0x10000000 ${filesize}; else echo NYX no nyx.bit; fi
nyxfdt=fdt resize 4096; fdt set /amba/spi@e0006000/ad9361-phy@0 compatible "adi,ad9361"; fdt set /amba/spi@e0006000/ad9361-phy@0 adi,2rx-2tx-mode-enable; fdt set /fpga-axi@0/cf-ad9361-dds-core-lpc@79024000 compatible "adi,axi-ad9361-dds-6.00.a"; fdt set /fpga-axi@0/mwipcore@43c00000 reg <0x43c70000 0x10000>; fdt mknode /fpga-axi@0 dma@40400000; fdt set /fpga-axi@0/dma@40400000 compatible "generic-uio"; fdt set /fpga-axi@0/dma@40400000 reg <0x40400000 0x10000>; fdt set /fpga-axi@0/dma@40400000 interrupts <0 29 4>; fdt mknode /fpga-axi@0 capbuf@1f000000; fdt set /fpga-axi@0/capbuf@1f000000 compatible "generic-uio"; fdt set /fpga-axi@0/capbuf@1f000000 reg <0x1f000000 0x1000000>
nyxrd=if fatload mmc 0 ${ramdisk_load_address} uramdisk.image.gz; then setenv nyxrdarg ${ramdisk_load_address}; else setenv nyxrdarg; echo NYX no uramdisk.image.gz - dung ramdisk QSPI; fi
nyxboot=run read_sf; run adi_loadvals; run qspiboot_extraenv; envversion; setenv bootargs console=ttyPS0,115200 maxcpus=${maxcpus} rootfstype=ramfs root=/dev/ram0 rw loglevel=4 mem=496M uio_pdrv_genirq.of_id=generic-uio; run nyxrd; bootm start ${fit_load_address}#${fit_config} ${nyxrdarg} && bootm loados && bootm ramdisk && bootm fdt && run nyxload && run nyxfdt && bootm prep && bootm go
bootcmd=echo NYX uEnv.txt active; run nyxboot; echo NYX staged boot FAILED - fallback to stock; run qspiboot
